void DeleteX(LinkList L, int x) {
    LNode *p = L->next;   // p 指向当前结点
    LNode *pre = L;        // pre 指向 p 的前驱
    while (p != NULL) {
        if (p->data == x) {
            pre->next = p->next;   // 断开 p
            free(p);                // 释放
            p = pre->next;          // p 移到下一个
        } else {
            pre = p;
            p = p->next;
        }
    }
}

// (2) 带头结点单链表就地逆置
void ReverseList(LinkList L) {
    LNode *p = L->next;   // p 指向第一个数据结点
    L->next = NULL;       // 头结点断开
    while (p != NULL) {
        LNode *q = p->next;   // 保存后继
        p->next = L->next;    // 头插法
        L->next = p;
        p = q;
    }
}

int CountNodes(BiTree T) {
    if (T == NULL)
        return 0;
    return CountNodes(T->lchild) + CountNodes(T->rchild) + 1;
}

// (2) 查找值为 x 的结点
BiTNode* SearchX(BiTree T, int x) {
    if (T == NULL)
        return NULL;
    if (T->data == x)
        return T;
    BiTNode *left = SearchX(T->lchild, x);
    if (left != NULL)
        return left;
    return SearchX(T->rchild, x);
}

```c
// 划分函数：以 A[low] 为枢轴，返回枢轴最终位置
int Partition(int A[], int low, int high) {
    int pivot = A[low];
    while (low < high) {
        while (low < high && A[high] >= pivot)
            high--;
        A[low] = A[high];
        while (low < high && A[low] <= pivot)
            low++;
        A[high] = A[low];
    }
    A[low] = pivot;
    return low;
}

// 快速排序递归函数
void QuickSort(int A[], int low, int high) {
    if (low < high) {
        int pivotpos = Partition(A, low, high);
        QuickSort(A, low, pivotpos - 1);
        QuickSort(A, pivotpos + 1, high);
}

// 调用入口
void QuickSortMain(int A[], int n) {
    QuickSort(A, 0, n - 1);
}
```

（2）时间复杂度分析：
- **最好情况**：每次划分均匀，递归树高度为 ⌊log₂n⌋，时间复杂度 O(n log₂n)。
- **最坏情况**：初始序列有序或逆序，每次划分极不均匀，递归树退化为单链，高度为 n，时间复杂度 O(n²)。
- **平均情况**：O(n log₂n)。
